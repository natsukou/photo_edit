========================================
手动部署命令模板
========================================

cd /path/to/project
git pull origin main
cd server
npm install
cp .env.example .env

# 编辑 .env，填写真实配置
# DASHSCOPE_API_KEY=YOUR_DASHSCOPE_API_KEY
# DB_HOST=YOUR_DB_HOST
# DB_PASSWORD=YOUR_DB_PASSWORD

pm2 restart photo-advice-server || pm2 start server.js --name photo-advice-server
pm2 logs photo-advice-server --lines 20

========================================
如果 PM2 进程不存在，执行以下命令启动
========================================

cd /path/to/project/server
pm2 start server.js --name photo-advice-server
pm2 save
pm2 startup

========================================
部署完成后测试
========================================

curl http://localhost:3000/api/ai/status
curl http://localhost:3000/health
